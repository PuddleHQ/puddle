fn fake_tool(dir: &Path, body: &str) -> PathBuf {
    let path = dir.join("tool");
    puddle_test_kit::write_executable(&path, &format!("#!/bin/sh\n{body}\n"));
    path
}

fn a_folder_is_not_a_script(dir: &Path) {
    std::fs::set_permissions(dir, std::fs::Permissions::from_mode(0o755)).unwrap();
    let data = dir.join("data");
    std::fs::write(&data, "x").unwrap();
    std::fs::set_permissions(&data, std::fs::Permissions::from_mode(0o644)).unwrap();
}

fn a_file_that_is_never_run(dir: &Path) {
    let file = dir.join("kept");
    std::fs::write(&file, "x").unwrap();
    // flake-ok: the test reads the mode back and never executes the file
    std::fs::set_permissions(&file, std::fs::Permissions::from_mode(0o755)).unwrap();
}
