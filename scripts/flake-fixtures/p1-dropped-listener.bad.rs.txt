// Original code, kept as it was; puddle-proxy tests/integration/pull.rs and puddle-e2e tests/integration/name_proxy.rs at 629fced84: a "dead" port made by binding port 0 and dropping the listener.
async fn a_dead_name_gets_a_502() {
    // A loopback port nothing listens on.
    let closed = TcpListener::bind((LOCAL, 0)).await.unwrap();
    let port = closed.local_addr().unwrap().port();
    drop(closed);
    let (head, _) = send(route.local_addr(), &connect(&format!("127.0.0.1:{port}"), Some(&auth))).await;
    assert!(head.starts_with("HTTP/1.1 502 "), "{head}");
}
